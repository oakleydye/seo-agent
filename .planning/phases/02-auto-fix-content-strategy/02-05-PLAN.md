```typescript
// ─── Phase 2: Fix PR formatters ──────────────────────────────────────────────

import type { Fix, FixResult, FixCategory, RiskCategory, FixPRTracking } from '../types/index.js';

const FIX_CATEGORY_DESCRIPTIONS: Record<FixCategory, string> = {
  'title-tag': 'Title tags appear as the clickable headline in search results and directly affect click-through rate (CTR). Missing or duplicate title tags reduce organic visibility.',
  'meta-description': 'Meta descriptions are shown as preview text in search results. They influence CTR and provide context to both users and search engines.',
  'og-tags': 'Open Graph tags control how pages appear when shared on social media. Missing tags result in unappealing auto-generated previews that reduce social traffic.',
  'alt-text': 'Image alt attributes improve accessibility (WCAG compliance) and give search engines context for image content, contributing to image search ranking.',
  'heading-hierarchy': 'Proper H1-H6 heading structure helps search engines understand page content hierarchy and improves accessibility for screen readers.',
  'schema-markup': 'Schema.org JSON-LD markup enables rich results in Google Search (star ratings, breadcrumbs, FAQs), increasing search result click-through rates.',
};

/**
 * Format the PR title for a fix run.
 * Example: "SEO Fix [meta-description]: acme-website — 2026-03-21"
 */
export function formatFixPRTitle(category: FixCategory, date: string, siteId: string): string {
  return `SEO Fix [${category}]: ${siteId} — ${date}`;
}

/**
 * Format the full PR body for a fix PR.
 * Includes: what was fixed, why, changes table, build validation status, discovery backlink.
 */
export function formatFixPRBody(fixResults: FixResult[], auditPRNumber?: number): string {
  const sections: string[] = [];

  // ── What Was Fixed ────────────────────────────────────────────────────────
  const categoryCounts = new Map<FixCategory, number>();
  for (const r of fixResults) {
    const count = categoryCounts.get(r.fix.category) ?? 0;
    categoryCounts.set(r.fix.category, count + 1);
  }

  const whatFixed = Array.from(categoryCounts.entries())
    .map(([cat, count]) => `- **${cat}**: ${count} file${count === 1 ? '' : 's'} fixed`)
    .join('\n');

  sections.push(`## What Was Fixed\n\n${whatFixed}`);

  // ── Why This Was Applied ──────────────────────────────────────────────────
  const categories = Array.from(new Set(fixResults.map(r => r.fix.category)));
  const whyLines = categories
    .map(cat => `**${cat}:** ${FIX_CATEGORY_DESCRIPTIONS[cat]}`)
    .join('\n\n');

  sections.push(`## Why This Was Applied\n\n${whyLines}`);

  // ── Changes Made ──────────────────────────────────────────────────────────
  const rows = fixResults.map(r => {
    const file = escapeTableCell(r.fix.sourceFile);
    const issueCount = r.fix.issueIds.length;
    const buildStatus = r.buildPassed ? '✓ Passed' : '✗ Failed';
    return `| ${file} | ${issueCount} | ${buildStatus} |`;
  });

  sections.push(`## Changes Made\n\n| File | Issues Fixed | Build Status |\n|------|-------------|-------------|\n${rows.join('\n')}`);

  // ── Build Validation ──────────────────────────────────────────────────────
  const allPassed = fixResults.every(r => r.buildPassed);
  const failedBuilds = fixResults.filter(r => !r.buildPassed);

  if (allPassed) {
    sections.push(`## Build Validation\n\n✓ All changes passed \`next build\` validation.`);
  } else {
    const errorDetails = failedBuilds
      .map(r => `**${r.fix.sourceFile}:** ${r.buildError ?? 'Unknown error'}`)
      .join('\n\n');
    sections.push(`## Build Validation\n\n⚠️ **Build failed on ${failedBuilds.length} file${failedBuilds.length === 1 ? '' : 's'}** — this PR is labeled \`needs-review\`. Please inspect the changes before merging.\n\n${errorDetails}`);
  }

  // ── Discovery ─────────────────────────────────────────────────────────────
  const discovery = auditPRNumber != null
    ? `Discovered by audit PR #${auditPRNumber}`
    : 'Run manually';

  sections.push(`## Discovery\n\n${discovery}`);

  // ── Footer ────────────────────────────────────────────────────────────────
  sections.push(`---\n*Generated by SEO Agent. Review changes and merge when satisfied.*`);

  return sections.join('\n\n');
}

/**
 * Get GitHub label names for a fix PR.
 * - 'seo-auto-fix': always applied
 * - 'needs-review': applied if risk is needs-review OR any fix failed build
 * - 'first-run': applied if submittedCount < firstRunLimit (trust establishment period)
 */
export function getFixPRLabels(
  riskCategory: RiskCategory,
  fixPRTracking: FixPRTracking,
  fixResults: FixResult[] = [],
): string[] {
  const labels: string[] = ['seo-auto-fix'];

  const buildFailed = fixResults.some(r => !r.buildPassed);
  if (riskCategory === 'needs-review' || buildFailed) {
    labels.push('needs-review');
  }

  if (fixPRTracking.submittedCount < fixPRTracking.firstRunLimit) {
    labels.push('first-run');
  }

  return labels;
}
```

```typescript
// ─── Phase 2: Fix PR creation ─────────────────────────────────────────────

import { formatFixPRTitle, formatFixPRBody, getFixPRLabels } from './pr-formatter.js';
import type { Fix, FixResult, FixCategory, RiskCategory, FixPRTracking } from '../types/index.js';

export interface FixPROptions {
  gitHubRepo: string;       // "owner/repo"
  category: FixCategory;
  fixResults: FixResult[];  // all fix results for this category PR
  siteId: string;           // for PR title
  date: string;             // YYYY-MM-DD
  auditPRNumber?: number;   // PR number from the audit that found these issues
  fixPRTracking: FixPRTracking; // for first-run label logic
  githubToken?: string;
}

/**
 * Create (or find existing) fix PR for a given fix category.
 * Branch naming: seo-fix/{category}/YYYY-MM-DD
 * Idempotent: returns existing PR if branch/PR already exists for today.
 * Applies labels: seo-auto-fix, needs-review (if applicable), first-run (if applicable).
 */
export async function createFixPR(options: FixPROptions): Promise<PRResult> {
  const { gitHubRepo, category, fixResults, siteId, date, auditPRNumber, fixPRTracking } = options;
  const token = options.githubToken ?? process.env['GITHUB_TOKEN'];

  if (!token) {
    throw new Error('GITHUB_TOKEN environment variable is required for PR creation');
  }

  const [owner, repo] = gitHubRepo.split('/');
  if (!owner || !repo) {
    throw new Error(`Invalid gitHubRepo format: "${gitHubRepo}" — expected "owner/repo"`);
  }

  const octokit = new Octokit({ auth: token });
  const branchName = `seo-fix/${category}/${date}`;
  const title = formatFixPRTitle(category, date, siteId);
  const body = formatFixPRBody(fixResults, auditPRNumber);
  const risk: RiskCategory = fixResults[0]?.fix.risk ?? 'needs-review';
  const labels = getFixPRLabels(risk, fixPRTracking, fixResults);

  // ─── Check for existing PR ───────────────────────────────────────────────
  const existingPRs = await octokit.pulls.list({
    owner,
    repo,
    state: 'open',
    head: `${owner}:${branchName}`,
    per_page: 1,
  });

  if (existingPRs.data.length > 0) {
    const existing = existingPRs.data[0]!;
    logger.info({ branchName, prUrl: existing.html_url }, 'Existing fix PR found — skipping creation');
    return {
      prUrl: existing.html_url,
      prNumber: existing.number,
      wasExisting: true,
    };
  }

  // ─── Get default branch SHA ───────────────────────────────────────────────
  const repoData = await octokit.repos.get({ owner, repo });
  const defaultBranch = repoData.data.default_branch;

  const defaultBranchData = await octokit.repos.getBranch({ owner, repo, branch: defaultBranch });
  const baseSha = defaultBranchData.data.commit.sha;

  // ─── Create branch if it doesn't exist ───────────────────────────────────
  let branchExists = false;
  try {
    await octokit.repos.getBranch({ owner, repo, branch: branchName });
    branchExists = true;
    logger.info({ branchName }, 'Fix branch already exists, skipping creation');
  } catch (err: unknown) {
    const octokitErr = err as { status?: number };
    if (octokitErr?.status !== 404) throw err;
  }

  if (!branchExists) {
    await octokit.git.createRef({
      owner,
      repo,
      ref: `refs/heads/${branchName}`,
      sha: baseSha,
    });
    logger.info({ branchName }, 'Fix branch created');
  }

  // ─── Ensure labels exist on the repo ─────────────────────────────────────
  const labelColors: Record<string, string> = {
    'seo-auto-fix': '0075ca',
    'needs-review': 'e4e669',
    'first-run': 'd93f0b',
  };

  for (const label of labels) {
    try {
      await octokit.issues.createLabel({
        owner,
        repo,
        name: label,
        color: labelColors[label] ?? 'ededed',
      });
    } catch (err: unknown) {
      // 422 = label already exists — expected, ignore
      const octokitErr = err as { status?: number };
      if (octokitErr?.status !== 422) {
        logger.warn({ label, error: (err as Error).message }, 'Failed to create label — continuing');
      }
    }
  }

  // ─── Create PR ────────────────────────────────────────────────────────────
  const prResponse = await octokit.pulls.create({
    owner,
    repo,
    title,
    body,
    head: branchName,
    base: defaultBranch,
  });

  // Apply labels
  if (labels.length > 0) {
    await octokit.issues.addLabels({
      owner,
      repo,
      issue_number: prResponse.data.number,
      labels,
    });
  }

  logger.info(
    { prUrl: prResponse.data.html_url, prNumber: prResponse.data.number, labels },
    'Fix PR created',
  );

  return {
    prUrl: prResponse.data.html_url,
    prNumber: prResponse.data.number,
    wasExisting: false,
  };
}
```
